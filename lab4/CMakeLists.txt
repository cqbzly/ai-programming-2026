cmake_minimum_required(VERSION 3.24)
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES native)
endif()
project(mytensor LANGUAGES CXX CUDA)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_STANDARD_REQUIRED ON)
find_package(Python REQUIRED COMPONENTS Interpreter Development.Module)
find_package(pybind11 CONFIG REQUIRED)
find_package(CUDAToolkit REQUIRED)

pybind11_add_module(_mytensor src/pybind.cpp src/tensor.cpp src/activation.cu src/lab3_kernels.cu src/ops.cu)
target_include_directories(_mytensor PRIVATE src)
set_target_properties(_mytensor PROPERTIES CUDA_RUNTIME_LIBRARY Static)
target_link_libraries(_mytensor PRIVATE CUDA::cudart_static CUDA::cublas)
if(MSVC)
    target_compile_options(_mytensor PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/utf-8>)
endif()
install(TARGETS _mytensor LIBRARY DESTINATION mytensor RUNTIME DESTINATION mytensor)
